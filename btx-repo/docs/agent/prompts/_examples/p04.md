## Wrong vs right

**Wrong** — wildcard allow
```rego
package btx.decisions
allow { true }
```

**Right** — conditioned allow with obligation
```rego
package btx.decisions
import future.keywords

decision := { "decision": "allow", "obligations": obs } {
  input.service_id == "income-cert.v3"
  input.purpose    == "scholarship_verification"
  data.grants[input.grant_id].active
  obs := [{ "shape": "yes_no_assertion", "ttl_seconds": 600 }]
}

test_allow_scholarship {
  decision == { "decision": "allow", "obligations":
    [{ "shape": "yes_no_assertion", "ttl_seconds": 600 }] }
    with input as { "service_id": "income-cert.v3",
                    "purpose": "scholarship_verification",
                    "grant_id": "g1" }
    with data.grants as { "g1": { "active": true } }
}
```
